// Render each test as a caught subtransaction, emitting structured diagnostics without normalizing JSON.
function render(t){return `\nreset role;${claims(t.actor)}${t.authenticated?"set local role authenticated;":""}\ndo $test$ declare r jsonb;q jsonb;code text;msg text;detail text;ctx text;name text;n bigint;begin\n begin\n${t.body}\nraise notice '%',jsonb_build_object('test',${lit(t.id)},'status','PASS','data',r,'summary',q,'observedSqlstate',code)::text;\nraise exception 'TEST_SUCCESS_SUBTRANSACTION_ROLLBACK' using errcode='Z0001';\nexception when sqlstate 'Z0001' then null;when others then get stacked diagnostics code=returned_sqlstate,msg=message_text,detail=pg_exception_detail,ctx=pg_exception_context;raise notice '%',jsonb_build_object('test',${lit(t.id)},'status','FAIL','sqlstate',code,'message',msg,'detail',detail,'context',ctx,'data',r,'summary',q)::text;end;end $test$;\nreset role;\n`;}
